src/simaws/iam.py, linter excerpt
def validate_policy_document(document: object) -> list[str]:
    """Return the list of least-privilege violations in a policy document.

    An empty list means the document is attachable. Every check exists to keep
    the simulator unable to express an over-broad grant.
    """
    errors: list[str] = []
    if not isinstance(document, dict):
        return ["policy document must be a JSON object"]
    if not isinstance(document.get("Version"), str):
        errors.append("policy document is missing a Version string")
    statements = document.get("Statement")
    if not isinstance(statements, list) or not statements:
        errors.append("Statement must be a non-empty list")
        return errors
    for index, statement in enumerate(statements):
        prefix = f"Statement[{index}]"
        if not isinstance(statement, dict):
            errors.append(f"{prefix} must be an object")
            continue
        if statement.get("Effect") != "Allow":
            errors.append(
                f"{prefix}: Effect must be 'Allow'; this linter only accepts least-privilege Allow statements"
            )
        errors.extend(_check_actions(prefix, statement.get("Action")))
        errors.extend(_check_resources(prefix, statement.get("Resource")))
        justification = statement.get("justification")
        if not isinstance(justification, str) or not justification.strip():
            errors.append(
                f"{prefix}: a non-empty 'justification' string is required for every statement"
            )
    return errors


def _check_actions(prefix: str, actions: object) -> list[str]:
    """Reject missing, non-string, or wildcarded action lists."""
    if not isinstance(actions, list) or not actions:
        return [f"{prefix}: Action must be a non-empty list of strings"]
    errors: list[str] = []
    for action in actions:
        if not isinstance(action, str):
            errors.append(f"{prefix}: Action entries must be strings")
        elif "*" in action:
            errors.append(
                f"{prefix}: wildcard action '{action}' rejected (least privilege)"
            )
        elif not _ACTION_RE.match(action):
            errors.append(f"{prefix}: action '{action}' must look like service:Action")
    return errors


def _check_resources(prefix: str, resources: object) -> list[str]:
    """Reject missing, non-string, or wildcarded resource lists."""
    if not isinstance(resources, list) or not resources:
        return [f"{prefix}: Resource must be a non-empty list of strings"]
    errors: list[str] = []
    for resource in resources:
        if not isinstance(resource, str):
            errors.append(f"{prefix}: Resource entries must be strings")
        elif "*" in resource or "?" in resource:
            errors.append(
                f"{prefix}: wildcard resource '{resource}' rejected (least privilege)"
            )
    return errors

"""Risk scoring agent with factor-sum validation.

Ported from the Week 8 multi-agent underwriting pipeline
(kenmann01/insurance-orchestrator at c0fd857), unchanged in behavior: the
model proposes a 0-100 score with a factor breakdown, code clamps the score
into range and rejects the output when the factor points do not actually add
up to the score (within a small tolerance), because an unexplained score is
exactly the kind of output an underwriter cannot trust.
"""

from riskservice.llm import AgentOutputError, LLMClient, complete_parsed
from riskservice.models import EnrichmentResult, IntakeResult, RiskResult

RISK_SYSTEM = (
    "You are the risk scoring agent. You score the underwriting risk of a "
    "case.\n"
    "You receive intake JSON and enrichment JSON.\n"
    'Return ONLY a JSON object of the form {"score": <int 0-100, higher is '
    'worse>, "factors": [{"name": ..., "points": <int>, "detail": ...}], '
    '"rationale": "..."}.\n'
    "Factor points MUST sum to the score."
)

SCORE_MIN = 0
SCORE_MAX = 100
FACTOR_SUM_TOLERANCE = 5


def risk_scoring_agent(
    intake: IntakeResult,
    enrichment: EnrichmentResult,
    *,
    llm: LLMClient,
) -> RiskResult:
    """Score the case with a 0-100 score and an explained factor breakdown.

    The model proposes score, factors, and rationale; code clamps the score
    into [SCORE_MIN, SCORE_MAX] and raises AgentOutputError when the factor
    points miss the clamped score by more than FACTOR_SUM_TOLERANCE, because
    an unexplained score is exactly the output an underwriter cannot trust.
    """
    user = (
        "Intake JSON:\n"
        + intake.model_dump_json()
        + "\n\nEnrichment JSON:\n"
        + enrichment.model_dump_json()
    )
    result = complete_parsed(llm, RISK_SYSTEM, user, RiskResult)

    score = max(SCORE_MIN, min(SCORE_MAX, int(result.score)))
    factors = list(result.factors) if result.factors else []
    total = sum(f.points for f in factors)
    if abs(total - score) > FACTOR_SUM_TOLERANCE:
        raise AgentOutputError(
            f"factor points sum {total} does not match score {score} "
            f"within tolerance {FACTOR_SUM_TOLERANCE}"
        )
    result.score = score
    result.factors = factors
    return result
src/riskservice/service.py
"""The deployable unit: score one case from recorded intake and enrichment
states. Both Week 9 deployment targets wrap exactly this function."""

import time

from riskservice.agent import risk_scoring_agent
from riskservice.llm import LLMClient, UsageRecordingClient
from riskservice.models import EnrichmentResult, IntakeResult, RiskResult


def run_risk_service(
    intake_data: dict,
    enrichment_data: dict,
    *,
    llm: LLMClient,
) -> tuple[RiskResult, dict]:
    """Score one case and report the token usage of the run.

    intake_data and enrichment_data are plain dicts matching IntakeResult and
    EnrichmentResult (the frozen input's keys). Returns the RiskResult plus a
    usage dict (input_tokens, output_tokens, model, duration_ms). Raises
    AgentOutputError when the model reply is invalid; never invents a score.
    """
    started = time.perf_counter()
    intake = IntakeResult.model_validate(intake_data)
    enrichment = EnrichmentResult.model_validate(enrichment_data)
    recording = UsageRecordingClient(llm)
    result = risk_scoring_agent(intake, enrichment, llm=recording)
    records = recording.drain()
    usage = {
        "input_tokens": sum(r.input_tokens for r in records),
        "output_tokens": sum(r.output_tokens for r in records),
        "model": records[-1].model if records else "",
        "duration_ms": round((time.perf_counter() - started) * 1000, 1),
    }
    return result, usage

class OllamaClient(LLMClient):
    """LLMClient backed by a local Ollama server's /api/chat endpoint.

    The transport parameter exists for tests: pass an httpx.MockTransport to
    intercept HTTP calls; in normal use it stays None and httpx uses its
    default transport. Host and model default to the environment variables
    RISKSERVICE_OLLAMA_HOST (or OLLAMA_HOST) and RISKSERVICE_MODEL, falling
    back to http://127.0.0.1:11434 and qwen3:8b.
    """

    # The timeout=240.0 and keep_alive="30m" defaults are deliberate: a cold
    # start can spend minutes reloading the model into memory, and the long
    # keep-alive keeps the model resident across repeated calls.
    def __init__(
        self,
        host: str | None = None,
        model: str | None = None,
        timeout: float = 240.0,
        temperature: float = 0.0,
        think: bool = False,
        keep_alive: str = "30m",
        transport: Any = None,
    ) -> None:
        """Store settings; a None host or model is resolved from the environment."""
        self.host = (host if host is not None else _default_host()).rstrip("/")
        self.model = model if model is not None else _default_model()
        self.timeout = timeout
        self.temperature = temperature
        self.think = think
        self.keep_alive = keep_alive
        self.transport = transport

    def _request(self, method: str, path: str, **kwargs: Any) -> httpx.Response:
        """Issue an HTTP request against the configured host."""
        # httpx's top-level helpers do not take a transport, so an injected
        # transport (tests) goes through a short-lived httpx.Client instead.
        if self.transport is not None:
            with httpx.Client(transport=self.transport) as http:
                return http.request(method, self.host + path, **kwargs)
        return httpx.request(method, self.host + path, **kwargs)

    def complete(self, system: str, user: str) -> LLMResponse:
        """POST /api/chat and map the reply into an LLMResponse.

        Reasoning blocks are stripped from the content, ollama's
        prompt_eval_count/eval_count become the token counts (0 when
        missing), and the model name falls back to the configured model
        when the server omits it.
        """
        payload = {
            "model": self.model,
            "messages": [
                {"role": "system", "content": system},
                {"role": "user", "content": user},
            ],
            "stream": False,
            "think": self.think,
            "keep_alive": self.keep_alive,
            "options": {"temperature": self.temperature},
        }
        response = self._request("POST", "/api/chat", json=payload, timeout=self.timeout)
        response.raise_for_status()
        data = response.json()
        message = data.get("message") or {}
        content = strip_think(message.get("content") or "")
        return LLMResponse(
            text=content,
            input_tokens=data.get("prompt_eval_count") or 0,
            output_tokens=data.get("eval_count") or 0,
            model=data.get("model") or self.model,
        )

    def health(self) -> bool:
        """True when the server answers GET /api/tags with a 200."""
        try:
            response = self._request(
                "GET", "/api/tags", timeout=HEALTH_CHECK_TIMEOUT_SECONDS
            )
            return response.status_code == 200
        except (httpx.HTTPError, OSError):
            # Transport-level failures only: a programming error must not
            # masquerade as "server unhealthy".
            return False

# Deployment Document: the Risk Scoring Agent

Lab: Deploy an Agent to AWS and a Simulated On-Prem Environment (Week 9).
This document covers the three documentation requirements of the brief: the
IAM role and why each permission exists, the networking setup and its
reasoning, and what changes architecturally between the cloud-style deployment
and the simulated on-prem deployment.

## The simulation, stated plainly

No cloud service is used or called anywhere in this lab. The "AWS" side runs
on `simaws`, a local, hand-rolled control plane built for this lab: roles and
policies in a JSON state store with a least-privilege linter, an image
registry with digest discipline, task definitions and a run-task verb that
genuinely enforces the role before executing anything, and a network placement
validator. ARNs use an `arn:simaws:` prefix. The "on-prem" side runs the same
service against a local Ollama server. The lab brief itself frames the second
target as a simulation of what a real air-gapped deployment looks like at
small scale, and this submission extends the same honesty to the first target.

## What is deployed

One agent from the Week 8 multi-agent underwriting pipeline: the risk scoring
agent (`risk_scoring_agent`), packaged standalone as the `riskservice` service
(`python -m riskservice`). It loads two recorded upstream states (intake and
enrichment), scores the case 0 to 100 with an explained factor breakdown,
refuses any output whose factor points do not sum to the score, and prints
the result with token usage. Both deployments run the same frozen test input:
the real intake and enrichment states that qwen3:8b produced for CASE-001
(Dana Whitfield) in the committed Week 8 trace, lifted verbatim into
`deploy/input/test_input.json`. That file also records the Week 8 reference
output (score 15, three 5-point factors) as a human reference; the service
never reads it.

The risk scoring agent is one of the two agents the brief suggests: it is
self-contained, and its factor-sum check gives the deployment a hard,
verifiable notion of "produces a correct result" for the same-input
confirmation.

## The IAM role and why each permission exists

Article 1 (AWS, Azure, GCP Fundamentals, Use Case 1) makes the principle
concrete with this very pipeline: the risk scoring agent "should run under a
role with exactly those two permissions... nothing more", because a
compromised or buggy credential with broad permissions "could cause damage far
beyond what the agent actually needs to do its job." That principle, least
privilege, is enforced by code here in two layers:

1. The policy linter (`simaws iam put-role-policy`) rejects any statement with
   a wildcard action, a wildcard resource, or a missing justification, so an
   over-broad policy cannot be attached at all, even by accident.
2. `run-task` authorizes before executing: the task's role must grant the
   image pull on that exact repository and log writes to that exact log group,
   or the task is denied and nothing runs (demonstrated by the denial tests
   and repeatable from the evidence transcripts).

The role `risk-service-task` carries one policy (`deploy/iam-policy.json`)
with three statements:

| Statement | Actions | Resource | Why each action is necessary |
|---|---|---|---|
| PullServiceImage | ecr:GetAuthorizationToken, ecr:BatchGetImage, ecr:GetDownloadUrlForLayer | the one `risk-service` repository | The task must pull exactly its own service image. GetAuthorizationToken authenticates the registry pull; BatchGetImage and GetDownloadUrlForLayer fetch manifest and layers. No push actions (a workload that can rewrite its own image source is a supply-chain risk), no other repository (lateral movement). |
| WriteTaskLogs | logs:CreateLogStream, logs:PutLogEvents | the one `/simaws/risk-service` log group | The task appends its own run output so the deployment can be audited. Read and delete stay with operators; a workload that can erase its own logs defeats the audit trail. |
| CallModelEndpoint | bedrock:InvokeModel | one specific model id | In a real AWS deployment the model call would be InvokeModel on one specific Bedrock model id, keeping model access under the same IAM model as everything else (article 1, Use Case 3). The simulator routes this call to the private Ollama endpoint gated by the security group egress rule instead, and the simulator does not enforce this statement; it is in the policy because it documents the real deployment shape. |

The role's trust policy allows assumption by the ECS task scheduler principal
only (`ecs-tasks.amazonaws.com`, the real service principal name), matching
the "roles are assumed by services, not hardcoded users" idea from article 1.

## Networking setup and reasoning

The placement lives in `deploy/network.json` and is validated by
`simaws ec2 validate-network` before anything deploys (article 1, Use Case 2:
internal services sit in private subnets and "should never be directly
reachable from outside"):

- VPC `vpc-0fde9001` (10.9.0.0/16), simulated.
- Three subnets: a public one reserved for a future entry point, and two
  private ones, application (10.9.1.0/24) and model (10.9.2.0/24).
- The task is placed in the **private application subnet**. The service is
  outbound-only: it receives no traffic, it scores a case and returns. A
  private subnet removes any direct internet path, and the validator rejects
  placement in a public subnet outright.
- Security group `sg-0fde9001risk`: **no ingress rules at all** (there is no
  inbound flow to allow, so none is opened; the validator rejects any
  world-open 0.0.0.0/0 ingress), and **exactly one egress rule**: tcp/11434 to
  the private model subnet (10.9.2.0/24), justified on the rule itself. The
  validator rejects any second egress rule, so the blast radius of a
  compromised task is one port to one subnet.
- The model endpoint sits in its own private subnet so that "the model runs
  inside our network boundary" is represented in the topology, the on-prem
  story in miniature.

The failure mode this guards against is the misconfigured-security-group
incident class: a task that can reach the whole internet can exfiltrate the
applicant data it scores. One egress rule to one internal subnet makes that
impossible by construction rather than by vigilance.

## The deployment flow (ECR + ECS path)

`scripts/deploy_aws_sim.py` runs the same sequence as the Week 3 console path,
in order, and its output is the deployment transcript:

1. validate the network placement;
2. create (or reuse) the task role, lint and attach the scoped policy;
3. create the repository, build and push the image (Docker build+save when a
   working Docker daemon is present, else a deterministic source archive with
   the same sha256 digest discipline);
4. register the task definition (image, role, environment), then run-task,
   which verifies the role's actions, the pushed digest, and the placement
   before executing anything;
5. print the task log: the service's scoring output on the frozen input.

## What changes architecturally for on-prem

Article 3 (On-Prem Realities, Use Case 3) answers this for the whole pipeline:
"the agent logic... stays conceptually identical; what changes is where the
model itself lives and how your code reaches it." Concretely in this lab:

- **What stays the same:** the service code, the agent, the factor-sum
  validation, the frozen input, the result schema. `riskservice` is byte-for-byte
  the same service in both deployments.
- **What changes:** only the model endpoint and model id. The cloud-style
  task calls a "managed endpoint" (qwen3:8b standing in, under the
  CallModelEndpoint statement's real-architecture story); the on-prem
  deployment calls the local Ollama server hosting a smaller open-weight
  model (qwen3:4b), selected by one flag. No code changes between the two:
  the endpoint is configuration, exactly the property article 3 describes.
- **What the air-gap would additionally demand** (article 3, Use Case 1):
  every dependency pre-staged through an approved transfer instead of pulled
  live (base images, Python packages, model weights), and updates moving at
  the client's audit pace, not at pip's. This lab's own equivalents are the
  committed artifacts: the image artifact with its recorded digest, the pinned
  model ids, and the frozen input.
- **Why a client demands it** (article 4): data residency and sovereignty
  (Use Cases 1 and 2: an inference call is a data transfer and can cross a
  jurisdiction), third-party vendor risk (Use Case 3: each external service
  triggers a formal review), and model risk management (Use Case 4: a
  self-hosted, version-pinned model can be frozen and validated; an external
  API can change underneath an audited decision). The on-prem variant here is
  version-pinned by model id, which is precisely the property article 4
  says regulated clients need.

# Comparison: simulated AWS deployment vs simulated on-prem deployment

The brief asks for three axes (cost, complexity, output quality) and one
specific recommendation: which deployment for a client with strict data
residency requirements, and which for one without. Both deployments ran the
same frozen test input (the Week 8 CASE-001 states scored by
`riskservice`); the run transcripts embedded as exhibits 3 and 6 carry the
measured outputs and the shared input hash.

## What was actually compared

- Cloud-style: the task runs under the least-privilege role in the simulated
  VPC's private application subnet; its model call goes to the
  managed-endpoint stand-in (qwen3:8b, the model id a real deployment would
  pin as its Bedrock model).
- On-prem-style: the same service on the same machine calls the boundary's
  local Ollama server hosting a deliberately smaller open-weight model
  (qwen3:4b).

Using two genuinely different model sizes is deliberate: the brief expects the
on-prem side to acknowledge "that a smaller open-weight model may reason
differently," and a same-model comparison would be identical by construction.

## Cost

Cloud-style (real-deployment shape, illustrative): an always-on Fargate task,
an ECR repository, a NAT path for the private subnet, plus per-token managed
model charges. The task itself is small, but the always-on footprint and the
managed model bill scale with traffic; at low, bursty volume you pay for idle
capacity. On-prem-style: hardware the client already owns inside the boundary,
zero per-token charges, and the marginal cost of a scoring call is
electricity. The trade is money for control: article 3's "real tradeoff"
framing, convenience versus control and compliance. The honest cost caveat on
the on-prem side is capacity engineering: serving many simultaneous
production users needs properly sized infrastructure (vLLM on server hardware,
article 3, Use Case 2), which is a capital line, not a subscription line.

## Complexity

Cloud-style carries the platform's complexity surface: IAM roles and policy
linting, registry and digest discipline, task definitions, subnets and
security groups. That complexity is exactly what this lab built, and it buys
auditability: every permission is written down, enforced, and evidenced
(exhibit 1). Operations are simpler though: the platform runs the task, keeps
the model warm and current, and scales the replica count.

On-prem-style is operationally simpler to reason about (one server, one model
file, no credentials anywhere) but pushes every platform concern onto the
owner: model hosting and upgrades, capacity, and the change-control process
from article 3 (Use Case 1), where even a dependency update moves at the
client's audit pace. The code side is a wash by design: the same service, one
endpoint flag different.

## Output quality

On the frozen input, the Week 8 recorded qwen3:8b reference (embedded in the
input file, produced at temperature 0) scored the case 15 with three
5-point factors and a clean rationale, using 218 input and 152 output tokens.
The run-day exhibits show both deployments' actual outputs on that same input
at temperature 0: the qwen3:8b task reproducing reference-quality scoring, and
the qwen3:4b on-prem model's attempt at the same task, with its factor-sum
validation outcome visible in the transcript. The structural guardrail
matters more than the model delta: whichever model serves, an output whose
factor points do not sum to its score is refused rather than recorded, so
quality degradation surfaces as visible refusals and repair calls, not as
quietly wrong scores. The general pattern to expect, consistent with the
course's Week 5 model-landscape material: the smaller model is faster and
free but drafts weaker factor breakdowns and needs its schema discipline
enforced by code.

## Recommendation

- **Client with strict data residency requirements: the on-prem deployment.**
  Not primarily for security hygiene, but because for such a client the cloud
  option is often not merely worse but non-compliant: an inference call is a
  data transfer (article 4, Use Case 2), each external service triggers a
  vendor-risk review that may end in "not approved" (Use Case 3), and model
  risk management wants a version-pinned model the client can freeze and
  validate (Use Case 4). The on-prem variant satisfies all three by
  construction: no external call exists, the model id is pinned, and the
  factor-sum gate provides the output auditability such clients need. At
  production scale, serve it with vLLM on sized hardware (article 3) rather
  than the development-grade Ollama used here.
- **Client without residency or sovereignty constraints: the cloud-style
  deployment.** It inherits the platform's access control, logging, and
  elasticity story, keeps the model operations with the provider, and its
  costs track usage instead of capital. The least-privilege role and the
  one-egress-rule placement travel with it, so the security posture does not
  depend on where the workload runs.

## Simulation caveat, restated

Neither side touched a real cloud or a real client data center: the "AWS" side
is a local control-plane simulation, and the "on-prem" side is a laptop-class
model server. What transfers to a real engagement is the architecture and its
enforcement: a scoped role with per-permission justifications, a
deny-by-default placement, a digest-pinned artifact, one frozen input
confirmed on both targets, and a deployment whose guarantees are checked by
code before anything executes.
